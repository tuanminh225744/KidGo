	\hline
	\textbf{Thuật ngữ}       & \textbf{Giải nghĩa}                                                          \\ \hline
	\textbf{Backend}         & Phần xử lý máy chủ và logic nghiệp vụ của hệ thống                           \\
	\textbf{Frontend}        & Phần giao diện tương tác trực tiếp với người dùng                            \\
	\textbf{In-memory Cache} & Bộ nhớ đệm lưu trữ dữ liệu trực tiếp trên RAM để truy xuất siêu tốc          \\
	\textbf{MERN Stack}      & Nền tảng công nghệ phát triển web gồm MongoDB, Express, React, Node.js       \\
	\textbf{Mocking}         & Kỹ thuật giả lập dữ liệu/phụ thuộc để phục vụ quá trình kiểm thử phần mềm    \\
	\textbf{Real-time}       & Quá trình truyền tải và xử lý dữ liệu với độ trễ cực thấp (thời gian thực)   \\
	\textbf{Ride-hailing}    & Dịch vụ gọi xe công nghệ thông qua nền tảng ứng dụng trực tuyến              \\
	\textbf{Responsive Web}  & Giao diện web có khả năng tự động điều chỉnh tương thích trên nhiều thiết bị \\
	\hline
\end{longtable}
